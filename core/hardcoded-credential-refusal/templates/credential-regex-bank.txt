# Credential regex bank for hardcoded-credential-refusal.
# Format: name<TAB>regex (ERE)<TAB>mode (optional: refuse|telemetry; default refuse).
# Lines starting with `#` are comments. Hand-edit to extend.
# The hook reads this file every invocation.
#
# `mode: telemetry` records the match in `~/.claude/telemetry.jsonl` with
# decision="telemetry-match" but does NOT exit 2. Used for high-FP regexes
# where the signal is useful but a refusal is too costly (e.g. git commit
# SHAs and file digests trip `high_entropy_hex` — see Finding-S4-1).
aws_access_key	AKIA[0-9A-Z]{16}
github_pat	gh[ps]_[A-Za-z0-9]{36}
stripe_live	sk_live_[0-9A-Za-z]{24}
openai_key	sk-[A-Za-z0-9]{48}
slack_bot	xoxb-[A-Za-z0-9-]+
jwt	eyJ[A-Za-z0-9_-]+\.eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+
rsa_private_key	-----BEGIN RSA PRIVATE KEY-----
# high_entropy_hex demoted to telemetry-only per Finding-S4-1: ERE has no
# look-behind, so we cannot exclude `commit <sha>` / `sha:<hash>` /
# `hash:<digest>` contexts inline. Telemetry-only preserves the signal
# without the velocity tax of refusing every state-changing edit on a
# hash-bearing file (README.md, CHANGELOG.md, lockfile entries, audit
# reports citing 40-hex commit SHAs).
high_entropy_hex	[a-f0-9]{40,}	mode: telemetry
